      });

        const pushX = Math.cos(angle) * intensity * 20;
        const pushY = Math.sin(angle) * intensity * 20;
        const rotate = (dx > 0 ? -1 : 1) * intensity * 10;
        span.style.transform = `translate(${pushX}px, ${pushY}px) rotate(${rotate}deg)`;
        const red = Math.floor(50 + (255 - 50) * intensity);
        const blue = Math.floor(70 + (255 - 70) * intensity);
        span.style.color = `rgb(${red}, 0, ${blue})`;
      } else {
        span.style.transform = "translate(0, 0) rotate(0deg)";
        span.style.color = "inherit";
      }
    });

    document.querySelectorAll('.question .word').forEach(word => {
      const rect = word.getBoundingClientRect();
      const wordX = rect.left + rect.width / 2;
      const wordY = rect.top + rect.height / 2;

      const dx = wordX - cursorX;
      const dy = wordY - cursorY;
      const distance = Math.sqrt(dx * dx + dy * dy);

      if (distance < 150) {
        const angle = Math.atan2(dy, dx);
        const intensity = (150 - distance) / 100;
        const pushX = Math.cos(angle) * intensity * 20;
        const pushY = Math.sin(angle) * intensity * 20;
        const rotate = (dx > 0 ? -1 : 1) * intensity * 5;
        word.style.transform = `translate(${pushX}px, ${pushY}px) rotate(${rotate}deg)`;
      } else {
        word.style.transform = "translate(0, 0) rotate(0deg)";
      }
    });
  }

  function handleEnd() {
    document.querySelectorAll('.question span').forEach(span => {
      span.style.transform = "translate(0, 0) rotate(0deg)";
      span.style.color = "inherit";
    });
    document.querySelectorAll('.question .word').forEach(word => {
      word.style.transform = "translate(0, 0) rotate(0deg)";
    });
  }
  document.addEventListener('touchend', handleEnd);

  const isTouchDevice = 'ontouchstart' in window || navigator.maxTouchPoints > 0;
  if (isTouchDevice) {
    document.addEventListener('touchmove', handleMove);
    document.addEventListener('touchend', handleEnd);
  } else {
    document.addEventListener('mousemove', handleMove);
    document.addEventListener('mouseup', handleEnd);
  }
  document.addEventListener('mousemove', handleMove);
  document.addEventListener('touchmove', handleMove);

  function loadQuestion() {
    const questionText = document.querySelector('.question');
    const optionsContainer = document.querySelector(".options");

    if (currentQuestion >= questions.length) {
      showResult();
      return;
    }

    questionText.innerHTML = '';
    optionsContainer.innerHTML = '';
    questionText.classList.remove('fade-in');

    questions[currentQuestion].split(' ').forEach(word => {
      const wordDiv = document.createElement('div');
      wordDiv.classList.add('word');
      word.split('').forEach(char => {
        const span = document.createElement('span');
        span.innerText = char;
        wordDiv.appendChild(span);
      });
      questionText.appendChild(wordDiv);
    });

    const questionIndex = currentQuestion;

    function playAnswerOption(index) {
      if (index >= options[questionIndex].length) return;

      const optionButton = document.createElement("button");
      optionButton.classList.add("option", "tuz");
      optionButton.innerText = options[questionIndex][index];
      optionButton.onclick = function () {
        selectAnswer(index + 1);
      };
      optionsContainer.appendChild(optionButton);
      addCursorEvents(optionButton);

      const answerAudio = new Audio(`voices/question${questionIndex + 1}_a${index + 1}.mp3`);
      answerAudio.play();
      answerAudio.onended = function () {
        playAnswerOption(index + 1);
      };
    }

    function showOptions() {
      if (soundEnabled) {
        playAnswerOption(0);
      } else {
        options[questionIndex].forEach((option, index) => {
          const optionButton = document.createElement("button");
          optionButton.classList.add("option", "tuz");
          optionButton.innerText = option;
          optionButton.onclick = function () {
            selectAnswer(index + 1);
          };
          optionsContainer.appendChild(optionButton);
          addCursorEvents(optionButton);
        });
      }
    }

    if (soundEnabled) {
      const questionAudio = new Audio(`voices/question${questionIndex + 1}.mp3`);
      questionAudio.play();
      questionAudio.onended = function () {
        showOptions();
      };
    } else {
      showOptions();
    }

    currentQuestion++;
    questionText.classList.add('fade-in');
  }

  function selectAnswer(answer) {
    answers.push(answer);
    const questionEl = document.querySelector('.question');
    questionEl.classList.add('fade-out');
    questionEl.addEventListener('animationend', function handler() {
      questionEl.classList.remove('fade-out');
      questionEl.removeEventListener('animationend', handler);
      loadQuestion();
    });
  }

function showResult() {
    let sum = answers.reduce((total, num) => total + num, 0);
    let outcome = (sum % 8) + 1;

 const dragonParts = {
      1: { body: 1, head: 1, legs: 1, wings: 1 },
      2: { body: 2, head: 2, legs: 2, wings: 2 },
      3: { body: 3, head: 3, legs: 3, wings: 3 },
      4: { body: 4, head: 4, legs: 4, wings: 4 },
      5: { body: 5, head: 5, legs: 5, wings: 5 },
      6: { body: 6, head: 6, legs: 6, wings: 6 },
      7: { body: 7, head: 7, legs: 7, wings: 7 },
      8: { body: 8, head: 8, legs: 8, wings: 8 },
    };
    let parts = dragonParts[outcome];

  let formData = new FormData();
    formData.append("outcome", outcome);
    formData.append("body", parts.body);
    formData.append("head", parts.head);
    formData.append("legs", parts.legs);
    formData.append("wings", parts.wings);

    fetch("save_dragon.php", {
      method: "POST",
      body: formData,
    })
      .then((response) => response.json())
      .then((data) => {
        let dragonId = data.dragon_id || "N/A";
        displayDragon(parts, dragonId);
      })
      .catch((error) => {
        console.error("Hiba történt az eredmény mentése során:", error);
        displayDragon(parts, "N/A");
      });
  }

          border-radius: 8px;

          padding: 5px;

      const dragonName = document.getElementById("dragonName").value.trim();
      const dragonColor = document.getElementById("dragonColor").value;
      if (!dragonName) {
        alert("Kérlek, add meg a sárkány nevét!");
        return;

      let formData = new FormData();
      formData.append("dragonName", dragonName);
      formData.append("color", dragonColor);
      formData.append("body", parts.body);
      formData.append("head", parts.head);
      formData.append("legs", parts.legs);
      formData.append("wings", parts.wings);
  
      fetch("save_dragon.php", {
        method: "POST",
        body: formData,
      })
        .then((response) => response.json())
        .then((data) => {

            alert("Hiba történt a sárkány mentésekor: " + data.error);
          } else {
            window.location.href = "index.php";
          }
        })
        .catch((error) => {

    const colorInput = document.getElementById("dragonColor");
    if (colorInput) {
      colorInput.addEventListener("input", function () {
        const hexColor = colorInput.value;

        let r = parseInt(hexColor.slice(1, 3), 16) / 255;
        let g = parseInt(hexColor.slice(3, 5), 16) / 255;
        let b = parseInt(hexColor.slice(5, 7), 16) / 255;

        const factor = 1.3;
        r = Math.min(r * factor, 1);
        g = Math.min(g * factor, 1);
        b = Math.min(b * factor, 1);
  
        const matrix = `
          ${r} 0 0 0 0
          0 ${g} 0 0 0
          0 0 ${b} 0 0
          0 0 0 1 0
        `.replace(/\s+/, ' ');
        
        const feColorMatrix = document.querySelector("#recolorFilter feColorMatrix");
        if (feColorMatrix) {
          feColorMatrix.setAttribute("values", matrix);
        }
      });
    }
